# Regenerates the aien-train rows of train/NOT-YET-OMEGA.tsv (2026-11-19). Run from a clone of aien-train
# holding f1bebdf3c5b08ec624005e5bc39d0427a87f9537; stdout is the exact block of aien-train rows.
# Rows: every path of `git ls-tree -r` at the commit except evidence/ and vendor/, *.om (imported under
# train/aien-train/ and train/omega-llm/) and docs/omegallm/*.md (imported under train/omega-llm/docs/).
# kind = text after the last dot of the file name (a leading dot does not count), else none.
# The same command on b9bafc3c5a450b44fc5dd544805e0a94479501ca with C=b9bafc3c5a45 and without the
# docs/omegallm exclusion reproduces the 1401 rows typed on 2026-11-18 (f197860c9) byte for byte.
# A pure-Omega train/rest/classify.om (like gpu/amd's) replacing this awk is a follow-up.
git ls-tree -r --name-only f1bebdf3c5b08ec624005e5bc39d0427a87f9537 \
| grep -v -E '^(evidence|vendor)/|\.om$|^docs/omegallm/[^/]+\.md$' \
| awk -v C=f1bebdf3c5b0 'BEGIN { OFS = "\t" }
{
  n = split($0, s, "/"); b = s[n]; t = substr(b, 2)
  if (t ~ /\./) { k = b; sub(/.*\./, "", k) } else k = "none"
  if (k == "sh" || k == "py") { c = "script-harness"; r = "shell/Python/JS/awk harness or tool; stays upstream until replaced by Omega" }
  else if (k == "md" || k == "svg" || k == "txt" || (k == "none" && b == "LICENSE")) { c = "doc-or-asset"; r = "documentation, static web asset, license or pin; stays upstream" }
  else if (k == "rs" || k == "toml" || k == "lock") { c = "rust-legacy"; r = "Rust source or manifest; stays upstream until rewritten in Omega" }
  else if (k == "c" || k == "cpp") { c = "c-native-legacy"; r = "C/C++/Swift/asm source or Makefile; stays upstream until rewritten in Omega" }
  else if (k == "none" && b == ".crumb") { c = "retired"; r = "retired crumb metadata (DRA-97); history only" }
  else { c = "data-or-fixture"; r = "fixture, test vector, config or receipt; stays upstream until its Omega consumer moves" }
  if ($0 == "tools/om_bundle_data.py" || $0 == "tools/om_compile_loop.py") r = "migrated host harness at tools/omega-llm/" b "; Python implementation"
  print "aien-train", C, $0, k, c, r
}'
