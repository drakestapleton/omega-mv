# ng-front prune script (seed/ng/front/prune.om): omc.om@943d9555 renamed by rename.map and hand-edited,
# minus the paths that cannot run in an ng front end.
#
# 1. Targets. fe_run exits 64 right after the lead lines for an x86/EFI target unit (D8) and for an IR v1
# unit, so f_xtarge and f_irmode are 0 in every unit the front end goes on with.
false ne get f_xtarge 0 0
true eq get f_xtarge 0 0
true ne get f_xtarge 0 1
false eq get f_xtarge 0 2
false eq get f_xtarge 0 1
false or ne get f_irmode 0 0 ne get f_xtarge 0 0
false and ne get f_xtarge 0 0 ne get f_om2 0 0
false and ne get f_xtarge 0 0 eq get f_scanni 0 0
true or eq get f_xtarge 0 0 eq get f_bdead i 0
false and eq w 6907493 eq get f_xtarge 0 2
false ne get f_irmode 0 0
call fe_xaddr
call fe_xstore
dead fe_run
#
# 2. Machine code. The check pass writes no program: the code buffer f_out, its offset f_oc and every
# global that only holds code offsets are sinks; the word writers and branch patchers go; then every
# function left without an effect goes with its call statements. An om2 fmt unit is checked (fmtprep) but
# not formatted: the fmtout calls under `if ne get f_fmtm 0 0` go, and f_fmtm is a sink.
sink f_out
sink f_oc
sink f_bbr
sink f_bhead
sink f_lastb
sink f_ilp
sink f_cpos
sink f_brkpos
sink f_fstart
sink f_fnaddr
sink f_callsi
sink f_tsite
sink f_tgoc
sink f_nt
sink f_tcode
sink f_fmtm
call fe_emit
call fe_emitb
call fe_patch19
call fe_patch26
call fe_blo
call fe_wrword
false ne get f_fmtm 0 0
false and ne get f_dead 0 0 gt n p0
pure
dead fe_run
tidy
pure
tidy
dead fe_run
