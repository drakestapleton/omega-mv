# ngbench v2 rows -> Wolfram data (reformatting only, no arithmetic):
#   d = <|{"case","arm"} -> <|block -> {wall1, wall2}, ..|>, ..|>   (measured blocks: warm = 0)
#   r = the same with INPROC_NS
#   cont = {{block, "case"}, ..}   (v3 K rows with contaminated=1 or busy=1; busyl: busy=1 only; contl: contaminated=1 only)
$1 == "T" && $3 == 0 {
	k = $4 "|" $5
	if (!(k in seen)) { seen[k] = 1; order[++nk] = k }
	kb = k SUBSEP $2
	if (!(kb in w)) { blist[k] = blist[k] " " $2 }
	w[kb] = (kb in w) ? w[kb] "," $7 : $7
	r[kb] = (kb in r) ? r[kb] "," $10 : $10
}
$1 == "K" && ($6 == "contaminated=1" || $7 == "busy=1") { cont = cont (cont == "" ? "" : ",") "{" $2 ",\"" $3 "\"}" }
$1 == "K" && $7 == "busy=1" { busyl = busyl (busyl == "" ? "" : ",") "{" $2 ",\"" $3 "\"}" }
$1 == "K" && $6 == "contaminated=1" { contl = contl (contl == "" ? "" : ",") "{" $2 ",\"" $3 "\"}" }
function emit(name, arr,   i, k, p, n, b, j, s) {
	printf "%s = <|", name
	for (i = 1; i <= nk; i++) {
		k = order[i]; split(k, p, "|")
		printf "%s{\"%s\",\"%s\"}-><|", (i > 1 ? "," : ""), p[1], p[2]
		n = split(blist[k], b, " "); s = ""
		for (j = 1; j <= n; j++) { printf "%s%s->{%s}", s, b[j], arr[k SUBSEP b[j]]; s = "," }
		printf "|>"
	}
	print "|>;"
}
END { emit("d", w); emit("r", r); print "cont = {" cont "};"; print "busyl = {" busyl "};"; print "contl = {" contl "};" }
