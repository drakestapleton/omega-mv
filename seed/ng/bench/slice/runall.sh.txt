#!/bin/bash
# ng-bench: R2 and S1, each waiting for an idle machine (scratch; evidence copy runall.sh.txt).
# Order X925 R2, X925 S1, then A725 R2, A725 S1: omp-queen-ng 14:15Z, no pool-10-11 lock from 14:28Z until
# omp-queen-spark's INF-4 END line, so the A725 rounds wait for that line on the BOARD (after line HOLD_LINE).
set -u
EV=~/work/ng/bench/slice/seed/ng/bench/evidence
TS=$(date -u +%Y%m%dT%H%MZ)
R2=$EV/$TS-r2; S1=$EV/$TS-s1
mkdir -p $R2 $S1
for d in $R2 $S1; do cp /tmp/ngb/sl/idleround.sh $d/idleround.sh.txt; cp /tmp/ngb/sl/runall.sh $d/runall.sh.txt; done
Q=~/handoffs/hive/queen
PA="omc ngA ngL ngB ngV ngM ngC ngR ngI ngF gccO2"; PC="bcopy arith branch isort sort leafcall matmul sieve linear"
SA="omc ngA ngS ngT gccO2"; SC="arith branch leafcall fib38 fib40 isort sieve sort matmul"
I=/tmp/ngb/sl/idleround.sh
$I 9 $Q/pool-8-9.lock 0 9 $R2/x925 "R2 payoff rerun" /tmp/ngb/payrun 31 3 "$PA" "$PC"
$I 9 $Q/pool-8-9.lock 0 9 $S1/x925 "S1 slice speed" /tmp/ngb/slb/fixture 31 1 "$SA" "$SC"
export HOLD_LINE=1891 HOLD_PAT='^[^ ]+ omp-queen-spark[^ ]*:? .*INF-4 (END|ENDED|DONE)' HOLD_NOT='until|expect|plan|will|ETA|by [0-9]'
$I 11 $Q/pool-10-11.lock 10 19 $R2/a725 "R2 payoff rerun" /tmp/ngb/payrun 31 3 "$PA" "$PC"
$I 11 $Q/pool-10-11.lock 10 19 $S1/a725 "S1 slice speed" /tmp/ngb/slb/fixture 31 1 "$SA" "$SC"
echo "$TS done" > /tmp/ngb/sl/runall.done
