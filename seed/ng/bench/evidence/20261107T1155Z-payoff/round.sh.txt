#!/bin/bash
# scratch round driver (not committed): round.sh CORE LOCK LO HI OUTDIR TAG
core=$1; lock=$2; lo=$3; hi=$4; E=$5; tag=$6
board() { flock ~/handoffs/BOARD.md.lock -c "echo \"$(date -u +%FT%TZ) ng-bench -> omp-queen-ng: $1\" >> ~/handoffs/BOARD.md"; }
date -u +%FT%TZ > $E/lock-request.txt
flock -x -w 3600 $lock bash -c "
board() { flock ~/handoffs/BOARD.md.lock -c \"echo \\\"\$(date -u +%FT%TZ) ng-bench -> omp-queen-ng: \$1\\\" >> ~/handoffs/BOARD.md\"; }
board 'LOCK HELD $(basename $lock) core $core: payoff round $tag (ngbench v2 31+3 blocks, 11 arms x 9 cases), about 10 min.'
date -u +%FT%TZ > $E/start.txt; cat /proc/loadavg >> $E/start.txt
ps -eo pid,psr,pcpu,stat,comm | awk 'NR==1 || (\$2>=$lo && \$2<=$hi)' > $E/cluster-start.txt
cat /sys/devices/system/cpu/cpu$core/regs/identification/midr_el1 > $E/midr.txt 2>&1
cd /tmp/ngb/payrun && taskset -c $core nice -n 10 /tmp/ngb/ngbench.run run 31 3 /tmp/ngb/payrun omc ngA ngL ngB ngV ngM ngC ngR ngI ngF gccO2 -- bcopy arith branch isort sort leafcall matmul sieve linear > $E/rows.txt 2> $E/stderr.txt
echo \$? > $E/rc.txt
date -u +%FT%TZ > $E/end.txt; cat /proc/loadavg >> $E/end.txt
ps -eo pid,psr,pcpu,stat,comm | awk 'NR==1 || (\$2>=$lo && \$2<=$hi)' > $E/cluster-end.txt
board \"RELEASED $(basename $lock) core $core: payoff round $tag done, rc=\$(cat $E/rc.txt), \$(grep '^S ' $E/rows.txt).\"
"
